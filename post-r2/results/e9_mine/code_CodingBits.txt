# Raw SLC, not the 8-bit preview - the preview crushes 30-37% of farm pixels onto one value.
def design(x_, y_): return np.column_stack([np.ones_like(x_), x_, y_, x_*x_, x_*y_, y_*y_])
def blk(a, valid, K):
    h, w = (a.shape[0]//K)*K, (a.shape[1]//K)*K
    s = np.where(valid, a, 0)[:h, :w].reshape(h//K, K, w//K, K).sum(axis=(1, 3))
    n = valid[:h, :w].reshape(h//K, K, w//K, K).sum(axis=(1, 3)).astype(np.float32)
    with np.errstate(invalid="ignore", divide="ignore"):
        return np.where(n > 0, s/np.maximum(n, 1), 0.).astype(np.float32)
v84 = vill.to_crs(4326).total_bounds
cor = np.array([[a_, b_] for a_ in (v84[0]-.02, v84[2]+.02) for b_ in (v84[1]-.02, v84[3]+.02)])
RAW = {}
for k in DATES:
    t0 = time.time()
    with rasterio.open(SLC[k]) as ds:
        g, gcrs = ds.gcps
        Dg = design(np.array([p.x for p in g]), np.array([p.y for p in g]))
        cc = np.linalg.lstsq(Dg, np.array([p.col for p in g]), rcond=None)[0]
        rr = np.linalg.lstsq(Dg, np.array([p.row for p in g]), rcond=None)[0]
        P_ = design(cor[:, 0], cor[:, 1]); pc, pr = P_ @ cc, P_ @ rr
        c0, c1 = int(max(pc.min()-100, 0)), int(min(pc.max()+100, ds.width))
        r0, r1 = int(max(pr.min()-100, 0)), int(min(pr.max()+100, ds.height))
        amp = np.abs(ds.read(1, window=Window(c0, r0, c1-c0, r1-r0))).astype(np.float32)
    val = amp > 0
    sg = blk((SFc[k]*amp)**2, val, 2)*math.sin(math.radians(INC[k])); del amp, val
    g2 = [GroundControlPoint(row=(p.row-r0)/2, col=(p.col-c0)/2, x=p.x, y=p.y,
                             z=getattr(p, "z", 0.)) for p in g]
    dst = np.zeros((H, W), np.float32)
    # gcps= alone fits a 2nd-order polynomial. SRC_METHOD='GCP_TPS' interpolates every GCP;
    # METHOD='GCP_TPS' is accepted and silently IGNORED by GDAL.
    reproject(source=sg, destination=dst, gcps=g2, src_crs=gcrs, src_nodata=0.,
              dst_transform=T, dst_crs="EPSG:%d" % UTM, dst_nodata=0.,
              resampling=Resampling.average, SRC_METHOD="GCP_TPS")
    # Verify TPS took effect: reproject again forcing poly2 and require the two to differ.
    if k == DATES[0]:
        _chk = np.zeros((H, W), np.float32)
        reproject(source=sg, destination=_chk, gcps=g2, src_crs=gcrs, src_nodata=0.,
                  dst_transform=T, dst_crs="EPSG:%d" % UTM, dst_nodata=0.,
                  resampling=Resampling.average, MAX_GCP_ORDER="2")
        _d = np.abs(_chk-dst); _n = np.isfinite(_d) & (_chk > 0) & (dst > 0)
        TPS_OK = bool(_n.sum() and _d[_n].max() > 0)
        del _chk, _d, _n
    RAW[k] = np.where(dst > 0, dst, np.nan); del sg, dst
    print("   %-6s %4.1fs | valid %.1f%% | median %.2f dB"
          % (DLAB[k], time.time()-t0, 100*np.isfinite(RAW[k]).mean(), 10*np.log10(np.nanmedian(RAW[k]))))

# Median, never mean: the worst bright scatterer inflates its plot's mean by ~19 dB. Every
# feature is de-biased against log(area) - sigma0 climbs with plot size on all four dates.
def fstat(arr, lab=LAB, nf=NF):
    f = lab.ravel(); s = f > 0
    v = arr.ravel()[s]; i = f[s]
    ok = np.isfinite(v) & (v > 0); i, v = i[ok], v[ok].astype(np.float64)
    o = np.argsort(i, kind="stable"); i, v = i[o], v[o]
    n = np.bincount(i, minlength=nf); e = np.concatenate([[0], np.cumsum(n)])
    med = np.full(nf, np.nan); cv = np.full(nf, np.nan); mean = np.full(nf, np.nan)
    for j in np.nonzero(n)[0]:
        x = v[e[j]:e[j+1]]; med[j] = np.median(x); mean[j] = x.mean(); cv[j] = x.std()/x.mean()
    return med, cv, mean, n.astype(float)

ntot = np.bincount(LAB.ravel()[LAB.ravel() > 0], minlength=NF).astype(float)

def on_grid(href, T_, W_, H_, rs=Resampling.average, dt="float32"):
    with rasterio.open(href) as ds:
        bb = rasterio.transform.array_bounds(H_, W_, T_)
        if ds.crs.to_epsg() != UTM: bb = transform_bounds(UTM, ds.crs, *bb)
        w_ = win_from_bounds(*bb, ds.transform).round_offsets().round_lengths()
        src = ds.read(1, window=w_, boundless=True, fill_value=0).astype(dt)
        st, sc = ds.window_transform(w_), ds.crs
    dst = np.zeros((H_, W_), dt)
    reproject(source=src, destination=dst, src_transform=st, src_crs=sc, dst_transform=T_,
              dst_crs="EPSG:%d" % UTM, resampling=rs)
    return dst
def z10(arr, nf=NF):
    f = LAB10.ravel(); s = f > 0
    v = arr.ravel()[s]; i = f[s]; ok = np.isfinite(v); i, v = i[ok], v[ok].astype(np.float64)
    n = np.bincount(i, minlength=nf).astype(float); sm = np.bincount(i, weights=v, minlength=nf)
    with np.errstate(invalid="ignore", divide="ignore"):
        return np.where(n > 0, sm/np.maximum(n, 1), np.nan), n

def chip(arr, cx, cy, half=110):
    r0, c0 = int(cy-half), int(cx-half)
    r0 = max(0, min(arr.shape[0]-2*half, r0)); c0 = max(0, min(arr.shape[1]-2*half, c0))
    return arr[r0:r0+2*half, c0:c0+2*half], (c0, r0)

def show_db(ax, a, vmin=-26, vmax=-10, cmap="magma"):
    with np.errstate(divide="ignore", invalid="ignore"):

    im = ax.imshow(d, cmap=cmap, vmin=vmin, vmax=vmax, interpolation="nearest")
    mapaxes(ax); return im

def _zc_y(sr):
    out = pd.Series(index=Q.index, dtype=float)
    for c in CROPS:
        m = (Q.crop_type == c) & sr.notna()
        ref = m & Q.clean.fillna(False)
        if ref.sum() < 8: ref = m
        mu, sd = sr[ref].mean(), sr[ref].std()
        out[m] = (sr[m]-mu)/(sd if sd and np.isfinite(sd) else 1.)
    return out.fillna(0).clip(-2.5, 2.5)

# pin at an unmeasured edge.
def _drv(wh):
    return 100/(1+np.exp(-((1-wh)*S_auc + wh*S_health)))
